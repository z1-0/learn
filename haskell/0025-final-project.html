2. 数据模型

-- src/Note.hs
module Note where

import Data.Time (UTCTime, getCurrentTime)
import Data.Text (Text)

data Note = Note
  { noteId      :: Int
  , noteContent :: Text
  , noteCreated :: UTCTime
  } deriving (Show, Eq)

newNote :: Int -> Text -> IO Note
newNote id content = do
  now <- getCurrentTime
  return Note { noteId = id, noteContent = content, noteCreated = now }

3. 存储层

-- src/Store.hs
module Store where

import Note
import Data.Text (Text)
import qualified Data.Text as T
import qualified Data.Text.IO as T
import Data.List (sortOn)
import System.Directory (doesFileExist)

-- 简单的文件存储（JSON 格式）
storePath :: FilePath
storePath = "notes.json"

loadNotes :: IO [Note]
loadNotes = do
  exists <- doesFileExist storePath
  if not exists
    then return []
    else do
      content <- T.readFile storePath
      case decodeNotes content of
        Nothing  -> return []
        Just ns -> return ns

saveNotes :: [Note] -> IO ()
saveNotes notes = do
  let content = encodeNotes notes
  T.writeFile storePath content

addNote :: Text -> IO ()
addNote content = do
  notes <- loadNotes
  let nextId = if null notes then 1 else maximum (map noteId notes) + 1
  note <- newNote nextId content
  saveNotes (note : notes)
  putStrLn ("Added note #" ++ show nextId)

listNotes :: IO ()
listNotes = do
  notes <- loadNotes
  let sorted = sortOn (Down . noteId) notes
  mapM_ printNote sorted
  where
    printNote n = putStrLn $
      show (noteId n) ++ ": " ++ T.unpack (noteContent n)

deleteNote :: Int -> IO ()
deleteNote id = do
  notes <- loadNotes
  let filtered = filter (\n -> noteId n /= id) notes
  if length filtered == length notes
    then putStrLn ("Note #" ++ show id ++ " not found")
    else do
      saveNotes filtered
      putStrLn ("Deleted note #" ++ show id)

searchNotes :: Text -> IO ()
searchNotes keyword = do
  notes <- loadNotes
  let matches = filter (\n -> keyword `T.isInfixOf` noteContent n) notes
  if null matches
    then putStrLn "No matching notes found"
    else mapM_ (\n -> putStrLn $ show (noteId n) ++ ": " ++ T.unpack (noteContent n)) matches

4. CLI 解析

-- src/CLI.hs
module CLI where

import Data.Text (Text)
import qualified Data.Text as T

data Command
  = Add Text
  | List
  | Delete Int
  | Search Text
  | Help

parseArgs :: [String] -> Command
parseArgs ["add", content] = Add (T.pack content)
parseArgs ["list"]         = List
parseArgs ["delete", idStr] =
  case reads idStr of
    [(id, _)] -> Delete id
    _         -> Help
parseArgs ["search", kw]   = Search (T.pack kw)
parseArgs _                = Help

showHelp :: String
showHelp = unlines
  [ "Note - Simple note-taking CLI"
  , ""
  , "Usage:"
  , "  note add     Add a new note"
  , "  note list             List all notes"
  , "  note delete       Delete a note"
  , "  note search  Search notes"
  ]

5. 入口文件

-- app/Main.hs
module Main where

import CLI
import Store
import System.Environment (getArgs)

main :: IO ()
main = do
  args <- getArgs
  case parseArgs args of
    Help        -> putStrLn showHelp
    Add content -> addNote content
    List        -> listNotes
    Delete id   -> deleteNote id
    Search kw   -> searchNotes kw

7. 扩展思路
