真实世界中，你经常需要同时使用多个 Monad 的效果（IO + Maybe + State）。Monad Transformer 让你组合多个 Monad 的能力。

1. 问题：Monad 不组合

-- 只有 IO：可以
readConfig :: IO String

-- 只有 Maybe：可以
parseConfig :: String -> Maybe Config

-- 但 IO + Maybe 呢？
-- 不能直接组合 IO 和 Maybe！
process :: IO (Maybe Config)  -- 也不好用

2. MaybeT：IO + Maybe

import Control.Monad.Trans.Maybe
import Control.Monad.Trans.Class (lift)

-- MaybeT IO 是一个 Monad，同时有 IO 和 Maybe 的效果
-- IO   的失败效果
-- Maybe 的失败效果

readConfig :: MaybeT IO Config
readConfig = do
  content <- lift (readFile "config.json")
  case parseConfig content of
    Nothing -> do
      lift (putStrLn "Parse failed!")
      MaybeT (return Nothing)
    Just cfg -> return cfg

-- 使用
main :: IO ()
main = do
  result <- runMaybeT readConfig
  case result of
    Nothing -> putStrLn "Failed to read config"
    Just cfg -> putStrLn ("Config: " ++ show cfg)

4. ReaderT：依赖注入

import Control.Monad.Trans.Reader

data Config = Config { dbUrl :: String, port :: Int }

-- ReaderT Config IO 可以访问 Config
getPort :: ReaderT Config IO Int
getPort = asks port

connectDB :: ReaderT Config IO ()
connectDB = do
  url <- asks dbUrl
  lift $ putStrLn ("Connecting to " ++ url)

-- 运行
main :: IO ()
main = do
  let config = Config "localhost" 8080
  runReaderT connectDB config

5. StateT：可变状态

import Control.Monad.Trans.State

-- StateT s IO a
-- s: 状态类型, a: 返回值

counter :: StateT Int IO ()
counter = do
  -- 修改状态
  modify (+1)
  -- 读取状态
  current <- get
  lift $ putStrLn ("Count: " ++ show current)

-- 运行
main :: IO ()
main = do
  -- evalStateT: 返回最终值
  -- execStateT: 返回最终状态
  -- runStateT: 返回 (值, 状态)
  result <- evalStateT (sequence_ (replicate 5 counter)) 0
  return ()

6. lift：提升操作

-- lift 将内层 Monad 的操作提升到 Transformer 中

-- 在 MaybeT IO 中使用 IO 操作
example :: MaybeT IO ()
example = do
  lift (putStrLn "Hello")     -- IO → MaybeT IO
  x <- lift getLine
  guard (x /= "")             -- Maybe 效果
  return ()

-- 多层提升
-- 在 StateT (MaybeT IO) 中

7. 实际例子：完整程序

{-# LANGUAGE GeneralizedNewtypeDeriving #-}

import Control.Monad.Trans.Reader
import Control.Monad.Trans.State
import Control.Monad.Trans.Class

data Config = Config { maxAttempts :: Int }
data GameState = GameState { attempts :: Int, secret :: Int }

-- 组合 App = ReaderT Config (StateT GameState IO)
type App = ReaderT Config (StateT GameState IO)

guess :: Int -> App Bool
guess n = do
  cfg <- ask
  st  <- get
  when (attempts st >= maxAttempts cfg) $
    liftIO $ putStrLn "No more attempts!"
  if n == secret st
    then return True
    else do
      put (st { attempts = attempts st + 1 })
      return False

runApp :: Config -> GameState -> App a -> IO a
runApp cfg st app = evalStateT (runReaderT app cfg) st

小结

  • Monad Transformer 组合多个 Monad 的效果
  • lift 提升内层 Monad 的操作
  • 常见：ReaderT（配置）、StateT（状态）、MaybeT（失败）
  • runXxxT 执行并解包
  • Transformer 堆栈是 Haskell 中管理复杂效果的惯用方式
